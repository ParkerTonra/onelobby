```ts
import type { Lobby } from "./lobby.js";
/**
 * A WebSocket-shaped view of one peer link, so code written against `WebSocket` runs on a lobby unchanged:
 * `lobby.asWebSocket()` on a joiner (the link to the host), `lobby.onConnection(conn => ...)` on the host (one per
 * joiner). Not a real WebSocket: no URL, no subprotocols, no `addEventListener`; `bufferedAmount` is always 0.
 * `D` is what `onmessage` gets: `Uint8Array` with `{ binary: true }`, `string` without.
 */
export interface LobbySocket<D extends string | Uint8Array = string | Uint8Array> {
    /** The peer at the other end (the host for a joiner's socket, the joiner for a host's connection). */
    readonly peerId: number;
    /** 0 CONNECTING (joiner, before `lobby.ready`), 1 OPEN, 3 CLOSED (there is no CLOSING). */
    readonly readyState: 0 | 1 | 3;
    readonly bufferedAmount: 0;
    /** A string is sent as UTF-8; bytes as they are. Ignored unless OPEN. */
    send(data: string | Uint8Array): void;
    /**
     * Closes this socket locally and fires `onclose` with `code` (default 1000). On a joiner it also leaves the lobby.
     * On the host it only stops this connection's events: to end the player's connection too (its socket gets your code
     * and reason), call `lobby.disconnect(conn.peerId, code, reason)`; `lobby.kick()` also bans it from the lobby.
     */
    close(code?: number, reason?: string): void;
    /**
     * Fires once, on a later task than the one that created the socket (like a real WebSocket). Assigned after it already
     * fired (the socket came out of an `await` chain)? It runs for the new handler once; `readyState === 1` also tells.
     */
    onopen: ((ev: {
        type: "open";
    }) => void) | null;
    /**
     * `data` is a string (UTF-8 decoded), or with `{ binary: true }` always a Uint8Array. It can be a view into a larger
     * buffer (check `byteOffset`/`byteLength`): `data.slice()` before transferring its `.buffer` to a Worker.
     */
    onmessage: MessageHandler<D> | null;
    /** `reason` is the lobby's reason ("host_left", "timeout", "kicked", "disconnected", ...) unless `close()` or the host gave one. */
    onclose: ((ev: {
        type: "close";
        code: number;
        reason: string;
        wasClean: boolean;
    }) => void) | null;
    /** Never called (errors surface as `onclose` with code 1006); present so WebSocket code that assigns it works. */
    onerror: ((ev: {
        type: "error";
    }) => void) | null;
}
/** Declared as a method so a `LobbySocket<Uint8Array>` still assigns to a plain `LobbySocket` (and back). */
type MessageHandler<D> = {
    handle(ev: {
        type: "message";
        data: D;
    }): void;
}["handle"];
export interface LobbySocketOptions {
    /** Deliver `onmessage` data as bytes (Uint8Array) instead of UTF-8 text. Default false. */
    binary?: boolean;
    /** Channel for `send` (default "reliable"). "fast" is unordered and may drop, like UDP. */
    channel?: "reliable" | "fast";
}
/**
 * Close codes: 1000 a normal leave, `close()` or the host's `disconnect()`, 1001 the other end went away (`host_left`,
 * the host closing, a player who left), 4000 kicked, 1006 anything abnormal (`timeout`, `ice_failed`, `server_lost`,
 * ...). A code and reason the host passed to `kick()` / `disconnect()` win over these.
 */
export declare function closeCodeFor(reason: string): number;
/**
 * Joiner: the link to the host as a socket. OPEN once `lobby.ready` resolves (at once if it already has), with
 * `onopen` on a later task;
 * closed when the session ends, or with 1001 "host_changed" when a server-backed lobby migrates to a new host.
 */
export declare function joinerSocket(lobby: Lobby, o?: LobbySocketOptions): LobbySocket;
/**
```

```ts
    udp?: () => UdpSocketLike | null;
}
/** What a relay probe finds out. */
export type RelayProbeResult = "ok" | "refused" | "unreachable" | "unknown";
export declare const CHECK_DEFAULT_STUN: readonly string[];
export declare function hexOf(b: Uint8Array): string;
export declare function bytesOf(hex: string): Uint8Array;
/** An RFC 5389 Binding Request with transaction id `tid` (12 bytes). */
export declare function stunBindingRequest(tid: Uint8Array): Uint8Array;
/** The mapped address in a Binding success response for `tid` (`XOR-MAPPED-ADDRESS`, else `MAPPED-ADDRESS`); null when there is none. */
export declare function parseStunResponse(bytes: Uint8Array, tid: Uint8Array): MappedAddress | null;
/** The `ip:port` a Binding Response for `tid` carries (an XOR-MAPPED-ADDRESS answer; the test STUN responders use it). */
export declare function stunBindingResponse(tid: Uint8Array, ip: string, port: number): Uint8Array;
/**
 * A TURN Allocate request for UDP (RFC 5766). Without `auth` it is the bare first request; with it, the long-term
 * credential form (USERNAME, REALM, NONCE, MESSAGE-INTEGRITY).
 */
export declare function turnAllocateRequest(tid: Uint8Array, auth?: {
    username: string;
    realm: string;
    nonce: Uint8Array;
    password: string;
}): Uint8Array | null;
/** What a TURN server answered: success, or an error code with the realm and nonce it wants (for the 401 round). */
export declare function parseTurnResponse(bytes: Uint8Array, tid: Uint8Array): {
    ok: true;
} | {
    ok: false;
    code: number;
    realm?: string;
    nonce?: Uint8Array;
} | null;
/** The slice of a UDP socket the check uses (node:dgram wrapped; a Godot or .NET client has its own). */
export interface UdpSocketLike {
    /** Binds and resolves with the local port and the local addresses (to tell an open NAT from a cone one). */
    bind(address?: string): Promise<{
        port: number;
        addresses: string[];
    }>;
    send(bytes: Uint8Array, host: string, port: number): void;
    onMessage: ((bytes: Uint8Array) => void) | null;
    close(): void;
}
/** A UDP socket over node:dgram, or null where there is none (a browser). */
export declare function nodeUdp(): UdpSocketLike | null;
/** True when every target is a loopback address (spec/diagnostics.md s.8): such a check binds 127.0.0.1, not all interfaces. */
export declare function allLoopback(targets: {
    host: string;
}[]): boolean;
/** `host` and `port` of a `stun:` / `turn:` URL (UDP only: `stuns:`, `turns:` and a tcp transport give null). */
export declare function iceUrlTarget(url: string): {
    host: string;
    port: number;
} | null;
/**
 * Probes a relay entry on its own (a fresh UDP socket): `ok`, `refused` (the server answered and turned the login down),
 * `unreachable` (no answer) or `unknown` (this runtime cannot test it: no UDP socket, a `turns:` URL). A lobby uses it to
 * tell `relay_refused` from `relay_unreachable` after a failed connect.
 */
export declare function probeRelay(s: IceServer, udp?: () => UdpSocketLike | null, ms?: number): Promise<RelayProbeResult>;
/**
 * Checks what this network allows: whether UDP gets out, the NAT's behaviour, and whether a configured relay takes the
 * login. Nothing is contacted but the STUN servers (default: the ones the client connects with) and the player's own
 * TURN entry. Run it from a menu; the library never runs it by itself.
 */
export declare function checkConnection(options?: CheckOptions): Promise<CheckResult>;
```
