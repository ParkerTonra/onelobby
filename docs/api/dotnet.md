## OneLobby.Replication/ReplicationSession.cs

```csharp
/// Host-authoritative state replication over a raw-mode Lobby (DESIGN s.18, spec/replication.md). Host: Spawn entities, run one sim step per NetworkClock.TryAdvance, read each client's input with TryGetInput, then SendSnapshot. Each client gets a delta against the last snapshot it acknowledged, on the fast channel. Client: SubmitInput once per Clock tick, render remote entities with TrySample{T} at RenderTick (100 ms behind the host), and reconcile the local player with TryGetLatest{T} + LastProcessedInput. Call Pump once per frame instead of lobby.Pump(); events fire inside it.
public sealed class ReplicationSession : IDisposable
    public ReplicationSession(Lobby lobby, ReplicationConfig? config = null);
    public ReplicationConfig Config { get; }
    public bool IsHost { get; }
    public Lobby Lobby;
    public int LocalPeerId;
    /// The host's peer id (same as Lobby.HostPeerId).
    public int HostPeerId;
    /// Client: round trip to the host in ms (OneLobby's 1 s pings; 0 until the first pong). Host: 0.
    public int RttMs;
    /// Round trip to peerId in ms (host: any client; client: only the host answers).
    public int PeerRttMs(int peerId);
    /// Host: the authoritative sim tick. Client: the local input tick (same rate, its own count).
    public NetworkClock Clock { get; }
    public ReplicationStats Stats { get; }
    /// Client: a reliable or fast event from the host; host: from a client. (from peerId, tick, payload).
    public event Action<int, uint, ReadOnlyMemory<byte>>? EventReceived;
    /// Every event, like EventReceived, with its stream and sequence number. Subscribe to one of the two.
    public event Action<ReplicatedEvent>? EventDelivered;
    /// A numbered stream skipped sequence numbers: (from, stream, first missing, next delivered), raised after the hole stayed open for StreamGapTimeoutMs. Reliable links never lose events, so this means the transport broke its contract; resync that stream's state (host: SendFullSnapshot plus your own facts).
    public event Action<int, byte, uint, uint>? EventGap;
    /// Client: an entity became visible at RenderTick (create the game object here).
    public event Action<EntityInfo>? EntitySpawned;
    /// Client: an entity is gone as of RenderTick (destroy the game object here).
    public event Action<EntityInfo>? EntityDespawned;
    /// Raw-mode payloads that are not replication messages (first byte other than 0x52), for the game's own use.
    public event Action<int, ReadOnlyMemory<byte>, Channel>? OtherMessage;
    /// Client: the host left. Host migration is not supported for replicated sessions (DESIGN s.18): end the match.
    public event Action? HostLost;
    /// True for payloads this session owns (first byte 0x52). Keep your own raw messages off that first byte.
    public static bool IsReplicationPayload(ReadOnlySpan<byte> payload);
    /// Pumps the lobby, then advances render time, raises spawn/despawn events and keeps acks flowing.
    public void Pump();
    /// Host: register an entity. owner is the controlling peerId, 0 = host-owned (AI, doors).
    public NetId Spawn(ushort typeId, int owner, IReplicatedEntity entity);
    /// Host: remove an entity; clients see it disappear at the snapshot tick it was last absent from.
    public bool Despawn(NetId id);
    public void SetOwner(NetId id, int owner);
    /// Host: the entities spawned with Spawn and not yet despawned, as (id, type id, owner peer id). A live view: to despawn while looping, copy it first (HostEntities.ToList()). Empty on a client: clients learn entities from EntitySpawned / EntityDespawned.
    public IEnumerable<EntityInfo> HostEntities;
    /// Host: peers receiving snapshots. A peer is added when its link is up (Lobby.PeerReady), not when the server announces it, and its first snapshot is a full one; it is removed on Lobby.PeerLeft.
    public IReadOnlyCollection<int> Clients;
    /// Host: true when peerId is in Clients (its link is up and it gets snapshots).
    public bool IsClientReady(int peerId);
    /// Host: a peer's link is up and it joined Clients (spawn its avatar here). Fires on the pumping thread.
    public event Action<int>? ClientReady;
    /// Host: newest snapshot tick this client acknowledged (0 = none yet: it gets full snapshots).
    public uint AckedTick(int peerId);
    /// Host: replication bytes sent to one client (payloads, no lobby header).
    public long BytesSentTo(int peerId);
    /// Host: one client's input counters (the per-peer split of ReplicationStats' Inputs* totals) and how many of its inputs are buffered right now. All zero for a peer that is not in Clients.
    public ClientInputStats InputStatsFor(int peerId);
    /// Host: the next snapshot this client gets is a full one (no baseline), and deltas resume only once it acknowledges a snapshot sent after this call. Use it when a client reports a gap or asks to resync.
    public void SendFullSnapshot(int peerId);
    /// Host: this tick's input from a client. One call per client per tick. Returns the next input in sequence; when none has arrived the previous one is repeated (InputFrame.Repeated); false only before the first.
    public bool TryGetInput(int peerId, out InputFrame input);
    /// Host: the tick to rewind hit volumes to for a shot in input, clamped to MaxRewindMs.
    public double RewindTick(in InputFrame input);
    /// Host: capture every entity's state at NetworkClock.Tick and send each client its delta. Call once at the end of each sim tick; ticks not on ReplicationConfig.SnapshotEveryTicks are skipped.
    public void SendSnapshot();
    /// Host: event to one client (toPeer) or all (0). Client: event to the host (toPeer ignored). Reliable by default (ordered, exactly once per link); Fast for cosmetic one-offs. Stamped with the current tick. stream 1-255 (reliable only) numbers the event per (stream, recipient) so the receiver sees ReplicatedEvent.Seq, drops duplicates and reports a skipped number with EventGap. Stream 0 (the default) is unnumbered.
    public void SendEvent(ReadOnlySpan<byte> payload, int toPeer = 0, Channel channel = Channel.Reliable, byte stream = 0);
    /// Reliable event bytes sent to one peer (payloads, no lobby header). Client: to the host.
    public long ReliableEventBytesSentTo(int peerId);
    /// Client: estimated current host tick (fractional).
    public double ServerTick;
    /// Client: the host tick being rendered: ServerTick minus the interpolation delay, smoothed.
    public double RenderTick;
    /// Client: newest snapshot tick received.
    public uint LatestTick;
    /// Client: ms since the newest snapshot (LatestTick) arrived; -1 before the first one.
    public long SnapshotAgeMs;
    /// Client: the last of my inputs the host had applied in the newest snapshot (for Reconciler{TInput,TState}).
    public uint LastProcessedInput;
    /// Client: entities visible at RenderTick.
    public IReadOnlyCollection<EntityInfo> Entities;
    public bool IsMine(EntityInfo e);
    /// Client: queue one input for the host and send it with the previous ReplicationConfig.InputRedundancy-1 inputs on the fast channel. Call once per Clock tick. Returns the input's sequence number.
    public uint SubmitInput(ReadOnlySpan<byte> input);
    /// Client: an entity's state interpolated at RenderTick. False if it is not visible then.
    public bool TrySample<T>(NetId id, out T state) where T : IReplicatedState<T>, new();
    /// Client: an entity's state interpolated at any tick still in the snapshot buffer.
    public bool TrySampleAt<T>(NetId id, double tick, out T state) where T : IReplicatedState<T>, new();
    /// Client: the entity's state in the newest snapshot (not interpolated), e.g. the local player's authoritative state.
    public bool TryGetLatest<T>(NetId id, out T state, out uint tick) where T : IReplicatedState<T>, new();
    /// Client: info for an entity in the newest snapshot (owner, type), even before it is visible.
    public EntityInfo? Find(NetId id);
    /// How long a numbered stream waits for a missing sequence number before EventGap skips it.
    public const int StreamGapTimeoutMs;
    public void Dispose();
```
